# -*- coding: utf-8 -*-
"""Инвентаризация и списания.

adjust_count — результат пересчёта: фактическое количество товара. Излишек
приходуется новым слоем по последней известной себестоимости (costing.last_cost),
недостача списывается по FIFO. Документ ADJ-xxxx, движение типа adjust с qty со
знаком (излишек +, недостача -).

writeoff — списание порчи/боя: количество уходит по FEFO (просроченные партии —
первыми) или из указанной партии lot, движение writeoff, документ WO-xxxx. Причина
обязательна (требование аудиторов).

Излишек при инвентаризации приходуется партией с кодом документа ADJ-xxxx без срока
годности; недостача списывается по FEFO, включая просроченные партии.
"""

from . import audit, catalog, costing, ids, inventory, validators
from .errors import ValidationError
from .models import record_movement
from .money import D, money_str, q2, qty_str


def adjust_count(db, sku, counted, date, reason=None):
    item = catalog.getItem(db, sku)
    sku = item["sku"]
    counted = validators.non_negative(counted, "фактическое количество")
    date = validators.date(date)
    before = inventory.on_hand(db, sku)
    diff = counted - before
    doc_id = ids.next_id(db, "adjust")
    cost = D(0)
    if diff > 0:
        unit_cost = costing.last_cost(db, sku)
        costing.add_layer(db, sku, doc_id, 1, date, diff, unit_cost, lot=doc_id)
        cost = q2(diff * unit_cost)
    elif diff < 0:
        allocations = costing.consume_fifo(db, sku, -diff)
        cost = -q2(costing.allocations_cost(allocations))
    if diff != 0:
        inventory.change_stock(db, sku, diff)
        record_movement(db, date, "adjust", sku, diff, cost, doc_id, note=reason)
    doc = {"id": doc_id, "type": "adjust", "date": date, "sku": sku, "before": qty_str(before),
           "after": qty_str(counted), "diff": qty_str(diff), "cost": money_str(cost)}
    if reason:
        doc["reason"] = reason
    db.data["documents"][doc_id] = doc
    audit.log(db, date, "adjust", doc=doc_id, sku=sku, diff=doc["diff"])
    return doc


def writeoff(db, sku, qty, date, reason, lot=None):
    item = catalog.getItem(db, sku)
    sku = item["sku"]
    qty = validators.positive_qty(qty)
    date = validators.date(date)
    if not reason or not reason.strip():
        raise ValidationError("укажите причину списания")
    allocations = costing.consume(db, sku, qty, lot=lot)
    inventory.change_stock(db, sku, -qty)
    cost = q2(costing.allocations_cost(allocations))
    doc_id = ids.next_id(db, "writeoff")
    record_movement(db, date, "writeoff", sku, qty, cost, doc_id, note=reason.strip())
    doc = {"id": doc_id, "type": "writeoff", "date": date, "sku": sku, "qty": qty_str(qty),
           "cost": money_str(cost), "reason": reason.strip(), "allocations": allocations}
    db.data["documents"][doc_id] = doc
    audit.log(db, date, "writeoff", doc=doc_id, sku=sku, qty=doc["qty"])
    return doc
