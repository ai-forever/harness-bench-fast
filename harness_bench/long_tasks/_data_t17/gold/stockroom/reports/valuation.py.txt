# -*- coding: utf-8 -*-
"""Оценка остатков по себестоимости партий.

Для каждого товара с ненулевым остатком: qty, value — сумма qty*cost по партиям,
округлённая до копеек, avg_cost — value / qty, округлённая до копеек. Итог — сумма
value по строкам. Цена партии — в рублях по курсу на дату прихода (историческая).

revalue=ДАТА — переоценка: валютные партии оцениваются по цене round2(cost_cur * курс
на эту дату), рублёвые — как обычно.
"""

from ..costing import layers_for, value_of
from ..inventory import on_hand
from ..money import D, fmt_money, fmt_qty, q2
from ..textable import render as _render

COLUMNS = [("sku", "Артикул"), ("name", "Наименование"), ("qty", "Остаток", "r"), ("avg_cost", "Ср. цена", "r"),
           ("value", "Стоимость", "r")]


def _value(db, sku, revalue=None):
    if revalue is None:
        return value_of(db, sku)
    from ..rates import get_rate

    total = D(0)
    for layer in layers_for(db, sku):
        unit = D(layer["cost"])
        if layer.get("currency"):
            unit = q2(D(layer["cost_cur"]) * get_rate(db, layer["currency"], revalue))
        total += D(layer["qty"]) * unit
    return total


def build(db, category=None, revalue=None):
    rows = []
    for sku in sorted(db.data["items"]):
        item = db.data["items"][sku]
        if category and item.get("category") != category:
            continue
        qty = on_hand(db, sku)
        if qty == 0:
            continue
        value = q2(_value(db, sku, revalue))
        rows.append({"sku": sku, "name": item["name"], "qty": qty, "avg_cost": q2(value / qty), "value": value})
    return rows


def total(db, category=None, revalue=None):
    return sum((r["value"] for r in build(db, category=category, revalue=revalue)), D(0))


def render(db, fmt="text", **opts):
    rows = build(db, **opts)
    if fmt == "csv":
        return _render(COLUMNS, rows, "csv")
    view = [dict(r, qty=fmt_qty(r["qty"]), avg_cost=fmt_money(r["avg_cost"]), value=fmt_money(r["value"]))
            for r in rows]
    foot = "Итого: %s" % fmt_money(sum((r["value"] for r in rows), D(0)))
    return _render(COLUMNS, view, "text", title="Оценка остатков", footer=foot)
