"""Командная строка.

    python3 -m stockroom [--db ПУТЬ] [--format text|csv] КОМАНДА [ПАРАМЕТРЫ]

Путь к базе: --db, иначе переменная окружения STOCKROOM_DB, иначе ./stockroom.json.
--format влияет только на отчёты и списки (report ..., item list, order list).

Коды возврата: 0 — успех, 2 — ошибка данных (сообщение в stderr начинается с
"ошибка: "), 1 — непредвиденная ошибка.

Каждая команда-обработчик получает (args, db) и возвращает текст для stdout.
Если обработчик изменил базу, main() сохраняет её (обработчики помечают это
через db.dirty = True).
"""

def cmd_item_show(args, db):
    item = catalog.getItem(db, args.sku)
    from .inventory import available, on_hand

    lines = ["%s — %s" % (item["sku"], item["name"]),
             "Единица: %s" % item.get("unit", "шт"),
             "Цена без НДС: %s" % fmt_money(item["price"]),
             "НДС: %s%%" % item["vat"],
             "Категория: %s" % (item.get("category") or "-"),
             "Штрихкод: %s" % (item.get("barcode") or "-"),
             "Остаток: %s" % fmt_qty(on_hand(db, item["sku"])),
             "Доступно: %s" % fmt_qty(available(db, item["sku"])),
             "Неснижаемый остаток: %s" % fmt_qty(item.get("min_qty", "0")),
             "Активен: %s" % ("да" if item.get("active", True) else "нет")]
    breaks = pricing.get_breaks(db, item["sku"])
    if breaks:
        lines.append("Скидки за количество: " + ", ".join("от %s — %s%%" % (fmt_qty(b["min_qty"]), fmt_qty(b["discount"]))
                                                      for b in breaks))
    lines.append("Валюта цены: %s" % (item.get("currency") or db.base_currency()))
    lines.append("Учёт сроков годности: %s" % ("да" if item.get("track_expiry") else "нет"))
    return "\n".join(lines)

REPORT_OPTIONS = {
    "stock": ("category", "include_inactive"),
    "low": (),
    "movements": ("date_from", "date_to", "sku", "mtype"),
    "valuation": ("category", "revalue"),
    "sales": ("date_from", "date_to", "group"),
    "abc": ("date_from", "date_to"),
    "turnover": ("date_from", "date_to"),
    "purchases": ("date_from", "date_to"),
    "debts": ("date_to",),
    "history": ("sku", "date_from", "date_to"),
    "dead": ("on_date", "days"),
    "margin": ("date_from", "date_to"),
    "supplier-prices": ("date_from", "date_to", "sku"),
    "sheet": ("category", "include_inactive"),
    "reorder": ("on_date", "days"),
    "writeoffs": ("date_from", "date_to"),
    "open-orders": (),
    "categories": (),
    "expiry": ("on_date", "days"),
}


def cmd_report(args, db):
    module = REPORTS[args.name]
    opts = {}
    for key in REPORT_OPTIONS[args.name]:
        value = getattr(args, key, None)
        if value not in (None, False):
            opts[key] = value
    if args.name == "turnover" and not (args.date_from and args.date_to):
        raise StockError("для оборачиваемости нужны --from и --to")
    if args.name == "history" and not args.sku:
        raise StockError("для карточки товара нужен --sku")
    if args.name == "dead" and not args.on_date:
        raise StockError("для отчёта о неликвидах нужна --on")
    if args.name == "reorder" and not args.on_date:
        raise StockError("для плана закупок нужна --on")
    if args.name == "expiry" and not args.on_date:
        raise StockError("для отчёта о сроках годности нужна --on")
    return module.render(db, args.format, **opts).rstrip("\n")


def cmd_export(args, db):
    if args.what == "items":
        n = csvio.export_items(db, args.file)
    elif args.what == "stock":
        n = csvio.export_stock(db, args.file)
    elif args.what == "lots":
        n = csvio.export_lots(db, args.file)
    else:
        n = csvio.export_movements(db, args.file, args.date_from, args.date_to)
    return "Выгружено строк: %d" % n


def cmd_import(args, db):
    if args.what == "items":
        rep = csvio.import_items(db, args.file, update=args.update)
        db.dirty = True
        out = ["Добавлено: %d, обновлено: %d, ошибок: %d" % (rep["added"], rep["updated"], len(rep["errors"]))]
        out.extend("  строка %d: %s" % e for e in rep["errors"])
        return "\n".join(out)
    if not args.date:
        raise StockError("для импорта остатков нужна --date")
    doc = csvio.import_stock(db, args.file, args.date)
    db.dirty = True
    return "Остатки загружены: приход %s, сумма %s" % (doc["id"], fmt_money(doc["total"]))

def cmd_stocktake(args, db):
    from . import stocktake

    if args.action == "start":
        session = stocktake.start(db, args.date, skus=args.skus or None, category=args.category)
        db.dirty = True
        return "Инвентаризация %s начата: позиций %d" % (session["id"], len(session["snapshot"]))
    if args.action == "count":
        if len(args.skus) != 2:
            raise StockError("использование: stocktake count SKU КОЛ-ВО")
        stocktake.record_count(db, args.skus[0], args.skus[1])
        db.dirty = True
        return "Учтено: %s = %s" % (args.skus[0].upper(), args.skus[1])
    if args.action == "diff":
        rows = stocktake.differences(db)
        return "\n".join("%s: учёт %s, факт %s, разница %s" % (r["sku"], fmt_qty(r["book"]), fmt_qty(r["counted"]),
                                                              fmt_qty(r["diff"])) for r in rows) or "пересчётов нет"
    if args.action == "post":
        docs = stocktake.post(db, args.date)
        db.dirty = True
        return "Инвентаризация проведена, документов: %d" % len(docs)
    stocktake.cancel(db)
    db.dirty = True
    return "Инвентаризация отменена"

def cmd_kit(args, db):
    from . import kits

    if args.action == "set":
        rows = kits.set_spec(db, args.sku, args.parts)
        db.dirty = True
        return "Спецификация %s: компонентов %d" % (args.sku.upper(), len(rows))
    if args.action == "show":
        spec = kits.get_spec(db, args.sku)
        lines = ["%s × %s" % (c["sku"], fmt_qty(c["qty"])) for c in spec]
        lines.append("Можно собрать: %d" % kits.can_assemble(db, args.sku))
        return "\n".join(lines)
    if not args.parts or len(args.parts) != 1:
        raise StockError("укажите количество комплектов")
    if args.action == "build":
        doc = kits.assemble(db, args.sku, args.parts[0], args.date)
        db.dirty = True
        return "Сборка %s: %s × %s, себестоимость единицы %s" % (doc["id"], doc["sku"], doc["qty"],
                                                                 fmt_money(doc["unit_cost"]))
    doc = kits.disassemble(db, args.sku, args.parts[0], args.date)
    db.dirty = True
    return "Разборка %s: %s × %s" % (doc["id"], doc["sku"], args.parts[0])

def _sub(parent, name, func, help_text):
    p = parent.add_parser(name, help=help_text)
    p.set_defaults(func=func)
    return p


def build_parser():
    ap = argparse.ArgumentParser(prog="stockroom", description="Складской учёт")
    ap.add_argument("--db", dest="db_path", default=None, help="путь к файлу базы")
    ap.add_argument("--format", choices=("text", "csv"), default="text", help="формат отчётов и списков")
    sub = ap.add_subparsers(dest="command")

    p = _sub(sub, "init", cmd_init, "создать новую базу")
    p.add_argument("--company", required=True)
    p.add_argument("--currency", default=config.BASE_CURRENCY)
    p.add_argument("--vat", default=config.DEFAULT_VAT)

    item = sub.add_parser("item", help="товары").add_subparsers(dest="item_cmd")
    p = _sub(item, "add", cmd_item_add, "добавить товар")
    p.add_argument("sku")
    p.add_argument("name")
    p.add_argument("--price", required=True)
    p.add_argument("--unit")
    p.add_argument("--vat")
    p.add_argument("--category")
    p.add_argument("--min-qty", dest="min_qty")

def main(argv=None):
    parser = build_parser()
    args = parser.parse_args(argv)
    if not getattr(args, "func", None):
        parser.print_help()
        return 1
    args.db_path = config.db_path(args.db_path)
    try:
        db = None if args.func is cmd_init else Database.open(args.db_path)
        out = args.func(args, db)
        if db is not None and db.dirty:
            db.save()
    except StockError as exc:
        sys.stderr.write("ошибка: %s\n" % exc)
        return 2
    if out:
        sys.stdout.write(out + "\n")
    return 0
