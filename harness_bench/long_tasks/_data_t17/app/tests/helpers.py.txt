# -*- coding: utf-8 -*-
"""Общие заготовки для тестов: стандартная база и запуск CLI."""

import os
import subprocess
import sys
import tempfile
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
FIXTURES = Path(__file__).resolve().parent / "fixtures"

from stockroom import catalog, partners, pricing, receiving  # noqa: E402
from stockroom.db import Database  # noqa: E402


def make_db(stocked=True):
    """База в памяти со справочниками:

    товары   MILK (шт, 89.90, НДС 10, мин. 20), BREAD (шт, 45.00, НДС 10, мин. 10),
             TEA (упак, 350.00, НДС 20, скидки 10:5 и 50:10), SUGAR (кг, 72.50, НДС 10)
    поставщики SUP1, SUP2
    покупатели C1 (розница, без скидки), C2 (опт, скидка 5%), C3 (розница, 3%)
    приходы (если stocked): R-0001 от 2024-03-01 SUP1: MILK 30@55.00, BREAD 40@28.00;
                            R-0002 от 2024-03-05 SUP2: MILK 20@58.00, TEA 100@210.00, SUGAR 25.5@48.00
    """
    db = Database.memory("ООО «Ромашка»")
    catalog.addItem(db, "MILK", "Молоко 1 л", "89.90", vat="10", min_qty=20, category="молочное", date="2024-01-10")
    catalog.addItem(db, "BREAD", "Хлеб белый", "45.00", vat="10", min_qty=10, category="хлеб", date="2024-01-10")
    catalog.addItem(db, "TEA", "Чай чёрный 100 пак.", "350.00", unit="упак", vat="20", category="бакалея",
                    date="2024-01-10")
    catalog.addItem(db, "SUGAR", "Сахар", "72.50", unit="кг", vat="10", category="бакалея", date="2024-01-10")
    pricing.set_breaks(db, "TEA", ["10:5", "50:10"])
    partners.add_supplier(db, "SUP1", "Молочный завод", inn="7701234567")
    partners.add_supplier(db, "SUP2", "Оптовая база")
    partners.add_customer(db, "C1", "Кафе «Уют»")
    partners.add_customer(db, "C2", "Сеть «Дом»", discount=5, group="wholesale")
    partners.add_customer(db, "C3", "Столовая № 5", discount=3)
    if stocked:
        receiving.receive(db, "SUP1", "2024-03-01", ["MILK:30@55.00", "BREAD:40@28.00"])
        receiving.receive(db, "SUP2", "2024-03-05", ["MILK:20@58.00", "TEA:100@210.00", "SUGAR:25.5@48.00"])
    return db


def save_db(db, directory):
    path = os.path.join(directory, "stock.json")
    db.save(path)
    db.path = path
    return path


def run_cli(db_path, *args):
    """Запустить `python -m stockroom --db db_path args...`; вернуть (код, stdout, stderr)."""
    env = dict(os.environ)
    env["PYTHONPATH"] = os.pathsep.join([str(ROOT)] + [p for p in env.get("PYTHONPATH", "").split(os.pathsep) if p])
    env["PYTHONIOENCODING"] = "utf-8"
    env.pop("STOCKROOM_DB", None)
    cmd = [sys.executable, "-m", "stockroom"]
    if db_path is not None:
        cmd += ["--db", str(db_path)]
    proc = subprocess.run(cmd + [str(a) for a in args], capture_output=True, text=True, encoding="utf-8",
                          env=env, cwd=tempfile.gettempdir())
    return proc.returncode, proc.stdout, proc.stderr
